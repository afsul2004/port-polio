def get_db() -> Generator[Session, None, None]:
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# Server-enforced Role-Based Access Control (RBAC)
def require_role(allowed_roles: list[str]):
    def role_checker(current_user = Depends(get_current_user)):
        if current_user.role not in allowed_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Access restricted for current user role"
            )
        return current_user
    return role_checker

# Automated eligibility calculation engine
def evaluate_eligibility(exam_score: float, aggregate_pct: float, job: JobPosition):
    """Evaluates candidate score against configurable job thresholds"""
    is_eligible = (
        exam_score >= job.min_exam_score and
        exam_score <= job.max_exam_score and
        aggregate_pct >= job.min_academic_percentage
    )
    decision = "ELIGIBLE" if is_eligible else "NOT_ELIGIBLE"
    return {
        "decision": decision,
        "score": exam_score,
        "academic": aggregate_pct
    }
